- **Textos e dinheiro:** textos de interface em português do Brasil. Dinheiro só via `formatBRL`.
- **Fronteiras de import:**
  - `src/app/` nunca acessa o banco; só módulos.
  - Módulos só via `@/modules/<nome>` ou `@/modules/<nome>/client`.
  - `../` é proibido em `src/` (exceto `schema.ts`).
  - Scripts (`scripts/`) importam só `@/modules/*` e `@/shared/*`.
- **Isolamento:** `storeId` sempre da sessão (`requireStoreAccess()`). Toda consulta de métricas e histórico filtra por `storeId`.
- **Período (spec §7):**
  - Hoje, 7 dias, 30 dias ou personalizado, sempre no fuso da loja (`store.timezone`).
  - Intervalo `[início do primeiro dia local, início do dia seguinte ao último)`.
  - 7 dias = hoje e os 6 anteriores; 30 dias = hoje e os 29 anteriores.
  - Personalizado: datas `YYYY-MM-DD`, início ≤ fim, no máximo 366 dias.
  - Valor desconhecido em `periodo` cai em **7 dias** (padrão das duas telas).
- **Indicadores (spec §7):**
  - Faturamento e ticket médio somam `total_cents` (com taxa de entrega) de todos os pedidos **exceto `cancelled`**.
  - "Pedidos" conta os não cancelados; "Cancelamentos" conta os `cancelled`.
  - Top 10, por hora, por tipo e por canal também ignoram cancelados.
  - Ticket médio = `round(faturamento / pedidos)`, ou 0 sem pedidos.
- **Top produtos:**
  - Item sem tamanho conta pelo `product_name`.
  - Pizza conta por sabor (`flavors`): meio a meio vale ½ unidade e metade do valor para cada sabor; tamanhos diferentes do mesmo sabor somam juntos.
  - Ordem: quantidade desc, faturamento desc, nome (`pt-BR`).
- **Hora do dia:** `extract(hour from created_at at time zone <fuso da loja>)`. `GROUP BY 1` (ordinal): o parâmetro do fuso não pode se repetir no `GROUP BY`.

```ts
export const PERIOD_PRESETS = ["hoje", "7d", "30d", "personalizado"] as const;
export type PeriodPreset = (typeof PERIOD_PRESETS)[number];

export const PERIOD_LABELS: Record<PeriodPreset, string> = {
  hoje: "Hoje",
  "7d": "Últimos 7 dias",
  "30d": "Últimos 30 dias",
  personalizado: "Personalizado",
};

export const DEFAULT_PERIOD: PeriodPreset = "7d";
export const MAX_PERIOD_DAYS = 366;

/** `from`/`to`: dias locais inclusivos (YYYY-MM-DD). `start`/`end`: instantes, `end` exclusivo. */
export type Period = { preset: PeriodPreset; from: string; to: string; start: Date; end: Date };
export type PeriodInput = { preset?: string | null; from?: string | null; to?: string | null };
export type PeriodResult = { ok: true; period: Period } | { ok: false; message: string };

const DAY_MS = 86_400_000;
const ISO_DATE = /^\d{4}-\d{2}-\d{2}$/;

function splitDate(isoDate: string): [number, number, number] {
  const [year = 0, month = 1, day = 1] = isoDate.split("-").map(Number);
  return [year, month, day];
}

function isIsoDate(value: string): boolean {
  if (!ISO_DATE.test(value)) return false;
  const [year, month, day] = splitDate(value);
  const date = new Date(Date.UTC(year, month - 1, day));
  return date.getUTCFullYear() === year && date.getUTCMonth() === month - 1 && date.getUTCDate() === day;
}

export function addDays(isoDate: string, days: number): string {
  const [year, month, day] = splitDate(isoDate);
  return new Date(Date.UTC(year, month - 1, day) + days * DAY_MS).toISOString().slice(0, 10);
}

function zonedParts(instant: Date, timeZone: string): Record<string, string> {
  const formatter = new Intl.DateTimeFormat("en-US", {
    timeZone,
    hourCycle: "h23",
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
    hour: "2-digit",
    minute: "2-digit",
    second: "2-digit",
  });
  return Object.fromEntries(formatter.formatToParts(instant).map((part) => [part.type, part.value]));
}

/** Diferença (ms) entre o relógio local do fuso e o UTC naquele instante. */
function offsetMs(instant: Date, timeZone: string): number {
  const p = zonedParts(instant, timeZone);
  const asUtc = Date.UTC(Number(p.year), Number(p.month) - 1, Number(p.day), Number(p.hour), Number(p.minute), Number(p.second));
  return asUtc - (instant.getTime() - instant.getUTCMilliseconds());
}

export function localDate(now: Date, timeZone: string): string {
  const p = zonedParts(now, timeZone);
  return `${p.year}-${p.month}-${p.day}`;
}

/** Instante da meia-noite local do dia `isoDate` no fuso (duas passadas cobrem a troca de horário de verão). */
export function zonedMidnight(isoDate: string, timeZone: string): Date {
  const [year, month, day] = splitDate(isoDate);
  const guess = Date.UTC(year, month - 1, day);
  const first = guess - offsetMs(new Date(guess), timeZone);
  return new Date(guess - offsetMs(new Date(first), timeZone));
}

function daySpan(from: string, to: string): number {
  return Math.round((Date.parse(to) - Date.parse(from)) / DAY_MS) + 1;
}

export function resolvePeriod(input: PeriodInput, timeZone: string, now: Date = new Date()): PeriodResult {
  const preset = (PERIOD_PRESETS as readonly string[]).includes(input.preset ?? "") ? (input.preset as PeriodPreset) : DEFAULT_PERIOD;
  const today = localDate(now, timeZone);
  let from: string;
  let to: string;

  if (preset === "personalizado") {
    from = input.from?.trim() ?? "";
    to = input.to?.trim() ?? "";
    if (!isIsoDate(from) || !isIsoDate(to)) return { ok: false, message: "Informe as datas de início e fim." };
    if (from > to) return { ok: false, message: "A data de início deve ser antes da data de fim." };
    if (daySpan(from, to) > MAX_PERIOD_DAYS) return { ok: false, message: `Escolha um período de até ${MAX_PERIOD_DAYS} dias.` };
  } else {
    to = today;
    from = preset === "hoje" ? today : addDays(today, preset === "7d" ? -6 : -29);
  }

  return {
    ok: true,
    period: { preset, from, to, start: zonedMidnight(from, timeZone), end: zonedMidnight(addDays(to, 1), timeZone) },
  };
}

function formatIsoDate(isoDate: string): string {
  const [year, month, day] = isoDate.split("-");
  return `${day}/${month}/${year}`;
}

export function formatPeriodRange(period: Pick<Period, "from" | "to">): string {
  if (period.from === period.to) return formatIsoDate(period.from);
  return `${formatIsoDate(period.from)} a ${formatIsoDate(period.to)}`;
}
```

`src/modules/orders/ranking.ts`:

```ts
export type RankableLine = { productName: string; sizeName: string | null; flavors: string[]; quantity: number; totalCents: number };
export type RankedProduct = { name: string; quantity: number; revenueCents: number };

/** Top produtos: pizza conta por sabor (meio a meio = ½ para cada); demais itens pelo nome. */
export function rankProducts(lines: RankableLine[], limit = 10): RankedProduct[] {
  const totals = new Map<string, { quantity: number; revenue: number }>();
  for (const line of lines) {
    const names = line.sizeName !== null && line.flavors.length > 0 ? line.flavors : [line.productName];
    for (const name of names) {
      const entry = totals.get(name) ?? { quantity: 0, revenue: 0 };
      entry.quantity += line.quantity / names.length;
      entry.revenue += line.totalCents / names.length;
      totals.set(name, entry);
    }
  }

  return [...totals]
    .map(([name, entry]) => ({ name, quantity: Math.round(entry.quantity * 100) / 100, revenueCents: Math.round(entry.revenue) }))
    .sort((a, b) => b.quantity - a.quantity || b.revenueCents - a.revenueCents || a.name.localeCompare(b.name, "pt-BR"))
    .slice(0, limit);
}

export function formatQuantity(quantity: number): string {
  return new Intl.NumberFormat("pt-BR", { maximumFractionDigits: 1 }).format(quantity);
}
```

```ts
import { and, asc, count, desc, eq, gte, lt, ne, sql } from "drizzle-orm";
import { db } from "@/shared/db/client";
import { rankProducts, type RankedProduct } from "./ranking";
import { orderItems, orders } from "./schema";
import type { Fulfillment, OrderChannel } from "./types";

export type StoreMetrics = {
  revenueCents: number;
  orderCount: number;
  averageTicketCents: number;
  cancelledCount: number;
  topProducts: RankedProduct[];
  byHour: { hour: number; orders: number }[];
  byFulfillment: { fulfillment: Fulfillment; orders: number; revenueCents: number }[];
  byChannel: { channel: OrderChannel; orders: number; revenueCents: number }[];
};

const revenue = sql<number>`coalesce(sum(${orders.totalCents}), 0)`.mapWith(Number);

/** Indicadores do período (spec §7): tudo exceto cancelados; cancelados só no contador próprio. */
export async function getStoreMetrics(storeId: string, range: { start: Date; end: Date }, timeZone: string): Promise<StoreMetrics> {
  const inRange = and(eq(orders.storeId, storeId), gte(orders.createdAt, range.start), lt(orders.createdAt, range.end));
  const valid = and(inRange, ne(orders.status, "cancelled"));
  // GROUP BY 1: o fuso vira parâmetro, e repetir a expressão no GROUP BY criaria outro parâmetro ($n) que o Postgres recusa.
  const localHour = sql<number>`extract(hour from ${orders.createdAt} at time zone ${timeZone})::int`.mapWith(Number);

  const [summaryRows, byHour, byFulfillment, byChannel, lines] = await Promise.all([
    db
      .select({
        orderCount: sql<number>`count(*) filter (where ${orders.status} <> 'cancelled')`.mapWith(Number),
        cancelledCount: sql<number>`count(*) filter (where ${orders.status} = 'cancelled')`.mapWith(Number),
        revenueCents: sql<number>`coalesce(sum(${orders.totalCents}) filter (where ${orders.status} <> 'cancelled'), 0)`.mapWith(Number),
      })
      .from(orders)
      .where(inRange),
    db.select({ hour: localHour, orders: count() }).from(orders).where(valid).groupBy(sql`1`).orderBy(sql`1`),
    db
      .select({ fulfillment: orders.fulfillment, orders: count(), revenueCents: revenue })
      .from(orders)
      .where(valid)
      .groupBy(orders.fulfillment)
      .orderBy(desc(count()), asc(orders.fulfillment)),
    db
      .select({ channel: orders.channel, orders: count(), revenueCents: revenue })
      .from(orders)
      .where(valid)
      .groupBy(orders.channel)
      .orderBy(desc(count()), asc(orders.channel)),
    db
      .select({
        productName: orderItems.productName,
        sizeName: orderItems.sizeName,
        flavors: orderItems.flavors,
        quantity: orderItems.quantity,
        totalCents: orderItems.totalCents,
      })
      .from(orderItems)
      .innerJoin(orders, eq(orderItems.orderId, orders.id))
      .where(valid),
  ]);

  const summary = summaryRows[0] ?? { orderCount: 0, cancelledCount: 0, revenueCents: 0 };
  return {
    revenueCents: summary.revenueCents,
    orderCount: summary.orderCount,
    averageTicketCents: summary.orderCount > 0 ? Math.round(summary.revenueCents / summary.orderCount) : 0,
    cancelledCount: summary.cancelledCount,
    topProducts: rankProducts(lines),
    byHour,
    byFulfillment,
    byChannel,
  };
}
```

```ts
  pageSize?: number;

export type OrderHistoryPage = { orders: QueueOrder[]; total: number; page: number; pageCount: number };

const ORDER_NUMBER = /^#?(\d{1,6})$/;
const PHONE_LIKE = /^[\d\s()+-]+$/;

function searchCondition(search: string | null): SQL | undefined {
  const term = search?.trim() ?? "";
  if (!term) return undefined;

  const numberMatch = ORDER_NUMBER.exec(term);
  if (numberMatch) return eq(orders.number, Number(numberMatch[1]));

  const digits = term.replace(/\D/g, "");
  if (PHONE_LIKE.test(term) && digits.length >= 8) return like(orders.customerPhone, `%${digits}%`);

  // Escapa curingas do LIKE: "%" e "_" digitados são texto, não padrão.
  return ilike(orders.customerName, `%${term.replace(/[\\%_]/g, "\\$&")}%`);
}

export async function listOrderHistory(storeId: string, filter: OrderHistoryFilter): Promise<OrderHistoryPage> {
  const pageSize = filter.pageSize ?? HISTORY_PAGE_SIZE;
  const where = and(
    eq(orders.storeId, storeId),
    gte(orders.createdAt, filter.start),
    lt(orders.createdAt, filter.end),
    filter.status ? eq(orders.status, filter.status) : undefined,
    searchCondition(filter.search),
  );

  const [counted] = await db.select({ total: count() }).from(orders).where(where);
  const total = counted?.total ?? 0;
  const pageCount = Math.max(1, Math.ceil(total / pageSize));
  const requested = Number.isFinite(filter.page) ? Math.floor(filter.page) : 1;
  const page = Math.min(Math.max(1, requested), pageCount);

  const rows = await db
    .select()
    .from(orders)
    .where(where)
    .orderBy(desc(orders.createdAt), desc(orders.number))
    .limit(pageSize)
    .offset((page - 1) * pageSize);

  return { orders: await hydrateOrders(rows), total, page, pageCount };
}
```

```tsx
function pageHref(period: Period, query: Query, page: number): string {
  const params = new URLSearchParams({ periodo: period.preset, de: period.from, ate: period.to });

  if (query.search) params.set("q", query.search);

  return `/painel/pedidos?${params.toString()}`;
```

```ts
import { randomBytes, randomUUID } from "node:crypto";
import { sql } from "drizzle-orm";
import { db } from "@/shared/db/client";
import type { PricedLine } from "./pricing";
import { orderCounters, orders, orderStatusEvents } from "./schema";
import { insertOrderLines } from "./service";
import type { Fulfillment, OrderAddress, OrderStatus, PaymentMethod } from "./types";

export type HistoricalOrder = {
  createdAt: Date;
  status: "completed" | "cancelled";
  cancelReason: string | null;
  fulfillment: Fulfillment;
  customerName: string;
  customerPhone: string;
  address: OrderAddress | null;
  tableLabel: string | null;
  paymentMethod: PaymentMethod;
  changeForCents: number | null;
  lines: PricedLine[];
  deliveryFeeCents: number;
};

const MINUTE = 60_000;

function statusTimeline(entry: HistoricalOrder): { status: OrderStatus; at: Date }[] {
  const at = (minutes: number) => new Date(entry.createdAt.getTime() + minutes * MINUTE);
  if (entry.status === "cancelled") {
    return [
      { status: "received", at: at(0) },
      { status: "cancelled", at: at(5) },
    ];
  }
  return [
    { status: "received", at: at(0) },
    { status: "preparing", at: at(3) },
    { status: entry.fulfillment === "delivery" ? "out_for_delivery" : "ready", at: at(30) },
    { status: "completed", at: at(45) },
  ];
}

/**
 * Só para seed/demonstração: grava pedidos antigos já finalizados, sem validar cardápio, horário nem preço.
 * Nunca usar em request de cliente.
 */
export async function importOrderHistory(storeId: string, history: HistoricalOrder[]): Promise<number> {
  if (process.env.NODE_ENV === "production") throw new Error("importOrderHistory não roda com NODE_ENV=production");
  const chronological = [...history].sort((a, b) => a.createdAt.getTime() - b.createdAt.getTime());

  for (const entry of chronological) {
    const timeline = statusTimeline(entry);
    const subtotalCents = entry.lines.reduce((sum, line) => sum + line.totalCents, 0);

    await db.transaction(async (tx) => {
      const [counter] = await tx
        .insert(orderCounters)
        .values({ storeId, lastNumber: 1 })
        .onConflictDoUpdate({ target: orderCounters.storeId, set: { lastNumber: sql`${orderCounters.lastNumber} + 1` } })
        .returning({ lastNumber: orderCounters.lastNumber });

      const [order] = await tx
        .insert(orders)
        .values({
          storeId,
          number: counter!.lastNumber,
          publicToken: randomBytes(32).toString("base64url"),
          idempotencyKey: randomUUID(),
          channel: "web",
          status: entry.status,
          fulfillment: entry.fulfillment,
          customerName: entry.customerName,
          customerPhone: entry.customerPhone,
          address: entry.fulfillment === "delivery" ? entry.address : null,
          tableLabel: entry.fulfillment === "dine_in" ? entry.tableLabel : null,
          paymentMethod: entry.paymentMethod,
          changeForCents: entry.paymentMethod === "cash" ? entry.changeForCents : null,
          notes: null,
          cancelReason: entry.status === "cancelled" ? entry.cancelReason : null,
          subtotalCents,
          deliveryFeeCents: entry.deliveryFeeCents,
          totalCents: subtotalCents + entry.deliveryFeeCents,
          createdAt: entry.createdAt,
          updatedAt: timeline[timeline.length - 1]!.at,
        })
        .returning({ id: orders.id });

      await insertOrderLines(tx, order!.id, entry.lines);
      await tx.insert(orderStatusEvents).values(timeline.map((event) => ({ orderId: order!.id, status: event.status, at: event.at })));
    });
  }
  return chronological.length;
}
```

```ts
import type { Menu, MenuProduct } from "@/modules/catalog";
import {
  addDays,
  localDate,
  priceCart,
  zonedMidnight,
  type CartItemInput,
  type Fulfillment,
  type HistoricalOrder,
  type PaymentMethod,
} from "@/modules/orders";

const HOUR = 3_600_000;
const MINUTE = 60_000;
const FIRST_DAY_AGO = 2; // nada nas últimas 24 h: a fila do painel e os E2E não veem o seed
const LAST_DAY_AGO = 30;
const MONDAY = 1; // demo fecha às segundas
const NAMES = [
  "Ana Souza", "Bruno Lima", "Carla Mendes", "Diego Alves", "Elisa Rocha", "Felipe Costa",
  "Gabriela Nunes", "Henrique Dias", "Isabela Martins", "João Pereira", "Larissa Gomes", "Marcos Ribeiro",
];
const STREETS = ["Rua das Palmeiras", "Av. Brasil", "Rua São Paulo", "Rua XV de Novembro", "Rua das Acácias"];
const NEIGHBORHOODS = ["Centro", "Jardim América", "Vila Nova", "Boa Vista", "São José"];
const CANCEL_REASONS = ["Cliente desistiu", "Endereço fora da área", "Sabor em falta"];
const EVENING_HOURS = [18, 19, 19, 20, 20, 20, 21, 21, 22, 23];

/** PRNG pequeno e determinístico (mulberry32): mesmo seed, mesmo histórico. */
function mulberry32(seed: number): () => number {
  let state = seed >>> 0;
  return () => {
    state = (state + 0x6d2b79f5) >>> 0;
    let t = state;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4_294_967_296;
  };
}

export type DemoHistoryOptions = { menu: Menu; timeZone: string; now: Date; deliveryFeeCents: number; seed?: number };

export function buildDemoHistory(options: DemoHistoryOptions): HistoricalOrder[] {
  const random = mulberry32(options.seed ?? 20261013);
  const pick = <T>(list: readonly T[]): T => list[Math.floor(random() * list.length)]!;
  const { menu } = options;
  const sizedCategories = menu.categories.filter((category) => category.sizes.length > 0);
  const simpleProducts = menu.categories.flatMap((category) =>
    category.sizes.length === 0 ? category.products.filter((product) => product.available && product.priceCents !== null) : [],
  );

  function pizza(): CartItemInput | null {
    if (sizedCategories.length === 0) return null;
    const category = pick(sizedCategories);
    const size = pick(category.sizes);
    const pool = category.products.filter((product) => product.available && product.sizePrices.some((price) => price.sizeId === size.id));
    if (pool.length === 0) return null;
    const halves = pool.filter((product) => product.allowsHalf);
    let flavors: MenuProduct[];
    if (size.maxFlavors >= 2 && halves.length >= 2 && random() < 0.35) {
      const first = pick(halves);
      flavors = [first, pick(halves.filter((product) => product !== first))];
    } else {
      flavors = [pick(pool)];
    }
    const groupIds = new Set(flavors.flatMap((product) => product.optionGroupIds));
    const groups = menu.optionGroups.filter((group) => groupIds.has(group.id) && group.minSelect === 0 && group.maxSelect >= 1);
    const optionIds: string[] = [];
    if (groups.length > 0 && random() < 0.25) {
      const options = pick(groups).options.filter((option) => option.available);
      if (options.length > 0) optionIds.push(pick(options).id);
    }
    return { kind: "sized", sizeId: size.id, flavorIds: flavors.map((product) => product.id), optionIds, quantity: 1, notes: null };
  }

  function simple(): CartItemInput | null {
    if (simpleProducts.length === 0) return null;
    return { kind: "simple", productId: pick(simpleProducts).id, optionIds: [], quantity: 1 + Math.floor(random() * 2), notes: null };
  }

  const today = localDate(options.now, options.timeZone);
  const history: HistoricalOrder[] = [];

  for (let daysAgo = LAST_DAY_AGO; daysAgo >= FIRST_DAY_AGO; daysAgo--) {
    const date = addDays(today, -daysAgo);
    const weekday = new Date(`${date}T00:00:00Z`).getUTCDay();
    if (weekday === MONDAY) continue;
    const midnight = zonedMidnight(date, options.timeZone).getTime();
    const ordersToday = 3 + Math.floor(random() * 5) + (weekday === 5 || weekday === 6 ? 4 : 0);

    for (let index = 0; index < ordersToday; index++) {
      const items: CartItemInput[] = [];
      const itemCount = 1 + Math.floor(random() * 3);
      for (let line = 0; line < itemCount; line++) {
        const item = line === 0 || random() < 0.4 ? pizza() : simple();
        if (item) items.push(item);
      }
      if (items.length === 0) continue;

      let lines;
      try {
        lines = priceCart({ menu, halfPriceRule: "highest", halfAndHalfEnabled: true, addonsEnabled: true }, items).lines;
      } catch {
        continue; // combinação que o cardápio não aceita: pula o pedido
      }

      const fulfillment: Fulfillment = random() < 0.65 ? "delivery" : "pickup";
      const deliveryFeeCents = fulfillment === "delivery" ? options.deliveryFeeCents : 0;
      const totalCents = lines.reduce((sum, line) => sum + line.totalCents, 0) + deliveryFeeCents;
      const paymentRoll = random();
      const paymentMethod: PaymentMethod = paymentRoll < 0.3 ? "cash" : paymentRoll < 0.7 ? "card_on_delivery" : "pix";
      const cancelled = random() < 0.07;

      history.push({
        createdAt: new Date(midnight + pick(EVENING_HOURS) * HOUR + Math.floor(random() * 60) * MINUTE),
        status: cancelled ? "cancelled" : "completed",
        cancelReason: cancelled ? pick(CANCEL_REASONS) : null,
        fulfillment,
        customerName: pick(NAMES),
        customerPhone: `+55119${String(Math.floor(random() * 100_000_000)).padStart(8, "0")}`,
        address:
          fulfillment === "delivery"
            ? {
                street: pick(STREETS),
                number: String(10 + Math.floor(random() * 990)),
                complement: null,
                neighborhood: pick(NEIGHBORHOODS),
                city: null,
                zipCode: null,
                reference: null,
              }
            : null,
        tableLabel: null,
        paymentMethod,
        changeForCents: paymentMethod === "cash" && random() < 0.5 ? Math.ceil((totalCents + 1) / 5000) * 5000 : null,
        lines,
        deliveryFeeCents,
      });
    }
  }
  return history;
}
```

```ts
import { ensureUser } from "@/modules/auth";
import { getMenu, importCatalog } from "@/modules/catalog";
import { importOrderHistory } from "@/modules/orders";
import { addStoreMember, createStore, deleteStoreBySlug, getStoreById, replaceStoreHours } from "@/modules/stores";
import { buildDemoHistory } from "./orders";
import { DEMO_OWNER, DEMO_STORE_SLUG, demoCatalog, demoHours, demoStore } from "./pizzaria-demo";

export async function seedDemoStore(): Promise<{ storeId: string; ownerUserId: string; historyCount: number }> {
  if (process.env.NODE_ENV === "production") throw new Error("seed:demo não roda com NODE_ENV=production");
  // Apagar em cascata e recriar mantém o seed idempotente.
  await deleteStoreBySlug(DEMO_STORE_SLUG);
  const { id: storeId } = await createStore(demoStore);
  await replaceStoreHours(storeId, demoHours);
  await importCatalog(storeId, demoCatalog);
  const owner = await ensureUser(DEMO_OWNER);
  await addStoreMember(storeId, owner.id);

  const store = await getStoreById(storeId);
  if (!store) throw new Error("loja demo não encontrada após criar");
  const deliveryFeeCents = store.rules.deliveryFee.strategy === "fixed" ? store.rules.deliveryFee.fixedCents : 0;
  const history = buildDemoHistory({ menu: await getMenu(storeId), timeZone: store.timezone, now: new Date(), deliveryFeeCents });
  const historyCount = await importOrderHistory(storeId, history);

  return { storeId, ownerUserId: owner.id, historyCount };
}
```
